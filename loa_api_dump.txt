  77      """Fire-and-forget webhook to n8n. Never blocks if n8n is down."""
  78      if extra is None:
  79          extra = {}
  80      url = os.getenv("N8N_WEBHOOK_URL", "")
  81      if not url:
  82          return
  83      try:
  84          async with httpx.AsyncClient(timeout=3.0) as client:

 249  def _apply_scope_filter(query, user: User):
 250      """Apply role-based scope filtering to a LOA query."""
 251      role = user.role_obj
 252      if role is None:
 253          return query.filter(False)
 254  
 255      if role.scope_type == 'cluster':
 256          query = query.filter(LOA.cluster.in_(role.scope_value or []))
 257      elif role.scope_type == 'brand':
 258          query = query.filter(LOA.brand.in_(role.scope_value or []))
 259      elif role.scope_type == 'function':
 260          query = query.filter(LOA.activity_type.in_(role.scope_value or []))
 261      elif role.scope_type == 'all':
 262          pass
 263      elif role.scope_type == 'author_only':
 264          query = query.filter(LOA.requestor_email == user.email)
 265      return query

 270  # If a LOA reaches global_review or global_functions_review but no active
 271  # reviewer is assigned for its brand/category, we skip the step automatically
 272  # instead of letting the LOA hang forever.
 273  
 274  def _has_reviewer_for_step(loa: LOA, target_status: str, db: Session) -> bool:
 275      """
 276      Returns True if at least one active user has comment rights at
 277      `target_status` AND a scope matching this LOA. False otherwise.
 278      """
 279      from backend.models.role import Role, RolePermission
 280      from backend.models.permission import Permission
 281  
 282      # Find all roles with `comment` permission at the target status.
 283      comment_perm = db.query(Permission).filter(Permission.code == "comment").first()
 284      if not comment_perm:
 285          return True  # safety: if permission system is misconfigured, do not skip
 286  
 287      roles_with_comment = (
 288          db.query(Role)
 289          .join(RolePermission, RolePermission.role_id == Role.id)
 290          .filter(RolePermission.permission_id == comment_perm.id)
 291          .filter(RolePermission.status == target_status)
 292          .all()
 293      )
 294  
 295      if not roles_with_comment:
 296          return False
 297  
 298      # For each candidate role, check if at least one user fits the LOA's scope
 299      for role in roles_with_comment:
 300          users_in_role = db.query(User).filter(User.role_id == role.id).all()
 301          if not users_in_role:
 302              continue
 303  
 304          # Scope check:
 305          #   - "all" scope ├ó┬å┬Æ any user matches
 306          #   - "brand" scope ├ó┬å┬Æ user.role.scope_value must include the LOA's brand
 307          #   - "function" scope ├ó┬å┬Æ user.role.scope_value must include a LOA category
 308          #   - "cluster" scope ├ó┬å┬Æ user.role.scope_value must include the LOA's cluster
 309          scope_type = role.scope_type
 310          scope_values = role.scope_value or []
 311  
 312          if scope_type == "all":
 313              return True
 314  
 315          if scope_type == "brand" and loa.brand:
 316              if any(loa.brand.lower() in (sv or "").lower() or (sv or "").lower() in loa.brand.lower() for sv in scope_values):
 317                  return True
 318  
 319          if scope_type == "function" and loa.category:
 320              cats = loa.category if isinstance(loa.category, list) else [loa.category]
 321              for cat in cats:
 322                  if any((cat or "").lower() in (sv or "").lower() or (sv or "").lower() in (cat or "").lower() for sv in scope_values):
 323                      return True
 324  
 325          if scope_type == "cluster" and loa.cluster:
 326              if any(loa.cluster.lower() == (sv or "").lower() for sv in scope_values):
 327                  return True
 328  
 329      return False

 332  # Status steps that we may skip if no reviewer is available
 333  SKIPPABLE_STATUSES = {"global_brand_review", "global_functions_review"}

 336  def _resolve_next_status(loa: LOA, current_status: str, db: Session) -> tuple[str, list[str]]:
 337      """
 338      Compute the next status, skipping orphan steps.
 339      Returns (final_status, list_of_skipped_steps).
 340      """
 341      skipped: list[str] = []
 342      next_status = WORKFLOW_TRANSITIONS.get(current_status)
 343  
 344      while next_status in SKIPPABLE_STATUSES:
 345          if _has_reviewer_for_step(loa, next_status, db):
 346              break  # someone can review here, stop and use this status
 347          skipped.append(next_status)
 348          next_status = WORKFLOW_TRANSITIONS.get(next_status)
 349          if next_status is None:
 350              break
 351  
 352      return next_status or "approved", skipped

 635  
 636  def _check_workflow_permission(user: User, loa: LOA, db: Session) -> str:
 637      current_status = loa.status
 638  
 639      if current_status in VALIDATION_STATUSES:
 640          required_permission = 'validate'
 641      elif current_status in ADVANCE_PHASE_STATUSES:
 642          required_permission = 'advance_phase'
 643      else:
 644          raise HTTPException(
 645              status_code=400,
 646              detail=f"Cannot advance a LOA in status '{current_status}'.",
 647          )
 648  
 649      role = user.role_obj
 650      if not role:
 651          raise HTTPException(status_code=403, detail="No role assigned")
 652  
 653      rp = (
 654          db.query(RolePermission, Permission)
 655          .join(Permission, RolePermission.permission_id == Permission.id)
 656          .filter(
 657              RolePermission.role_id == role.id,
 658              RolePermission.loa_status == current_status,
 659          )
 660          .first()
 661      )
 662      actual_permission = rp[1].code if rp else None
 663  
 664      if not actual_permission or not implies(actual_permission, required_permission):
 665          raise HTTPException(
 666              status_code=403,
 667              detail={
 668                  'error': 'Permission denied',
 669                  'required_permission': required_permission,
 670                  'user_role': role.code,
 671                  'loa_status': current_status,
 672                  'message': (
 673                      f"Your role '{role.code}' cannot perform "
 674                      f"'{required_permission}' on a LOA in status '{current_status}'."
 675                  ),
 676              },
 677          )
 678  
 679      return required_permission

 682  @router.post("/{loa_id}/validate")
 683  async def validate_workflow_step(
 684      loa_id: str,
 685      db: Session = Depends(get_db),
 686      user: User = Depends(get_current_user),
 687  ):
 688      loa = db.query(LOA).filter(LOA.id == loa_id).first()
 689      if not loa:
 690          raise HTTPException(status_code=404, detail="LOA not found")
 691  
 692      if loa.is_locked:
 693          raise HTTPException(
 694              status_code=423,
 695              detail="LOA is locked by a super-admin. It must be unlocked first.",
 696          )
 697  
 698      current_status = loa.status
 699      if current_status not in WORKFLOW_TRANSITIONS:
 700          raise HTTPException(
 701              status_code=400,
 702              detail=f"Cannot validate a LOA in status '{current_status}'. "
 703                     f"Valid statuses: {list(WORKFLOW_TRANSITIONS.keys())}",
 704          )
 705  
 706      permission_used = _check_workflow_permission(user, loa, db)
 707  
 708      # Double validation for CMO/CFO step
 709      if current_status == "cmo_cfo_review":
 710          role = user.role_obj
 711          role_code = role.code if role else ""
 712  
 713          if role_code == "regional_cmo":
 714              loa.cmo_approved = True
 715          elif role_code == "regional_cfo":
 716              loa.cfo_approved = True
 717          else:
 718              loa.cmo_approved = True
 719              loa.cfo_approved = True
 720  
 721          if loa.cmo_approved and loa.cfo_approved:

1047  os.makedirs(UPLOAD_DIR, exist_ok=True)

1050  @router.post("/{loa_id}/photos")
1051  async def upload_photo(
1052      loa_id: str,
1053      file: UploadFile = File(...),
1054      db: Session = Depends(get_db),
1055  ):
1056      loa = db.query(LOA).filter(LOA.id == loa_id).first()
1057      if not loa:
1058          raise HTTPException(status_code=404, detail="LOA not found")
1059  
1060      existing = [f for f in os.listdir(UPLOAD_DIR) if f.startswith(loa_id)]
1061      if len(existing) >= 2:
1062          raise HTTPException(
1063              status_code=400,
1064              detail=f"Maximum 2 photos per LOA. {len(existing)} already exist: {', '.join(existing)}. Delete existing photos first via DELETE /loa/{loa_id}/photos/1 or /2."
1065          )
1066  
1067      ext = file.filename.split(".")[-1].lower()
1068      if ext not in ["jpg", "jpeg", "png"]:
1069          raise HTTPException(status_code=400, detail="Only JPG and PNG allowed")
1070  
1071      photo_index = len(existing) + 1
1072      filename = f"{loa_id}_photo{photo_index}.{ext}"
1073      filepath = os.path.join(UPLOAD_DIR, filename)
1074  
1075      with open(filepath, "wb") as f:
1076          shutil.copyfileobj(file.file, f)
1077  
1078      return {"filename": filename, "url": f"/loa/{loa_id}/photos/{photo_index}"}

1081  @router.get("/{loa_id}/photos/{index}")
1082  async def get_photo(loa_id: str, index: int):
1083      for f in os.listdir(UPLOAD_DIR):
1084          if f.startswith(f"{loa_id}_photo{index}"):
1085              ext = f.rsplit(".", 1)[-1].lower()
1086              media_type = {
1087                  "jpg": "image/jpeg", "jpeg": "image/jpeg", "png": "image/png",
1088              }.get(ext, "application/octet-stream")
1089              return FileResponse(os.path.join(UPLOAD_DIR, f), media_type=media_type)
1090      raise HTTPException(status_code=404, detail="Photo not found")

1093  @router.delete("/{loa_id}/photos/{index}")
1094  async def delete_photo(loa_id: str, index: int):
1095      for f in os.listdir(UPLOAD_DIR):
1096          if f.startswith(f"{loa_id}_photo{index}"):
1097              os.remove(os.path.join(UPLOAD_DIR, f))
1098              return {"deleted": True}
1099      raise HTTPException(status_code=404, detail="Photo not found")
